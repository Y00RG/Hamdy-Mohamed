Sub Forma_Print()

    Dim wbBO As Workbook
    Dim wsTemplate As Worksheet
    Dim wsPrint As Worksheet
    Dim wsLME As Worksheet
    Dim wsCAR As Worksheet
    
    Dim LastRowLME As Long
    Dim LastRowCAR As Long
    Dim i As Long


    '========================================================
    ' WORKBOOKS / SHEETS
    '========================================================

    Set wbBO = Workbooks("BO.xlsx")
    Set wsTemplate = ThisWorkbook.Worksheets("BO Settled")
    
    Set wsLME = wbBO.Worksheets("LME")
    Set wsCAR = wbBO.Worksheets("CAR")


    '========================================================
    ' STEP 1 - Set today's date in the template
    '========================================================

    wsTemplate.Range("A1").Value = Date
    wsTemplate.Range("A1").NumberFormat = "d/m/yyyy"


    '========================================================
    ' STEP 2 - Copy BO Settled template to BO.xlsx
    '========================================================

    wsTemplate.Copy After:=wbBO.Sheets(3)
    
    Set wsPrint = wbBO.ActiveSheet


    '========================================================
    ' STEP 3 - Find actual last row in LME
    '========================================================

    LastRowLME = wsLME.Cells(wsLME.Rows.Count, "B").End(xlUp).Row


    '========================================================
    ' STEP 4 - Copy LME Name + Status
    ' B:C  --->  I:J
    '========================================================

    If LastRowLME >= 2 Then

        wsLME.Range("B2:C" & LastRowLME).Copy

        wsPrint.Range("I2").PasteSpecial _
            Paste:=xlPasteAllUsingSourceTheme, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

        wsPrint.Range("I2").PasteSpecial _
            Paste:=xlPasteValues, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

    End If


    '========================================================
    ' STEP 5 - Copy LME Branch + Rim No
    ' E:F  --->  K:L
    '========================================================

    If LastRowLME >= 2 Then

        wsLME.Range("E2:F" & LastRowLME).Copy

        wsPrint.Range("K2").PasteSpecial _
            Paste:=xlPasteAllUsingSourceTheme, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

        wsPrint.Range("K2").PasteSpecial _
            Paste:=xlPasteValues, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

    End If


    '========================================================
    ' STEP 6 - Clean LME Branch Names
    ' K = Branch
    '========================================================

    If LastRowLME >= 2 Then

        For i = 2 To LastRowLME

            If Len(Trim(wsPrint.Cells(i, "K").Value)) > 0 Then

                wsPrint.Cells(i, "K").Value = _
                    CleanBranch(wsPrint.Cells(i, "K").Value)

            End If

        Next i

    End If


    '========================================================
    ' STEP 7 - Generate LME numbering dynamically
    ' H = #
    '========================================================

    If LastRowLME >= 2 Then

        wsPrint.Range("H2:H" & LastRowLME).Value = _
            wsPrint.Evaluate("ROW(1:" & LastRowLME - 1 & ")")

    End If


    '========================================================
    ' STEP 8 - Format LME area
    '========================================================
    
    If LastRowLME >= 2 Then
    
        With wsPrint.Range("H2:L" & LastRowLME)
    
            With .Font
                .Name = "Arial"
                .Size = 18
                .Bold = True
                .Strikethrough = False
                .Superscript = False
                .Subscript = False
                .OutlineFont = False
                .Shadow = False
                .Underline = xlUnderlineStyleNone
                .Color = -16777216
                .TintAndShade = 0
                .ThemeFont = xlThemeFontNone
            End With
    
            With .Borders
                .LineStyle = xlContinuous
                .Weight = xlThin
            End With
    
        End With
    
    End If


    '========================================================
    ' STEP 9 - Find actual last row in CAR
    '========================================================

    LastRowCAR = wsCAR.Cells(wsCAR.Rows.Count, "B").End(xlUp).Row


    '========================================================
    ' STEP 10 - Copy CAR Name + Status
    ' B:C  --->  C:D
    '========================================================

    If LastRowCAR >= 2 Then

        wsCAR.Range("B2:C" & LastRowCAR).Copy

        wsPrint.Range("C2").PasteSpecial _
            Paste:=xlPasteAllUsingSourceTheme, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

        wsPrint.Range("C2").PasteSpecial _
            Paste:=xlPasteValues, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

    End If


    '========================================================
    ' STEP 11 - Copy CAR Branch + Rim No
    ' E:F  --->  E:F
    '========================================================

    If LastRowCAR >= 2 Then

        wsCAR.Range("E2:F" & LastRowCAR).Copy

        wsPrint.Range("E2").PasteSpecial _
            Paste:=xlPasteAllUsingSourceTheme, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

        wsPrint.Range("E2").PasteSpecial _
            Paste:=xlPasteValues, _
            Operation:=xlNone, _
            SkipBlanks:=False, _
            Transpose:=False

    End If


    '========================================================
    ' STEP 12 - Clean CAR Branch Names
    ' E = Branch
    '========================================================

    If LastRowCAR >= 2 Then

        For i = 2 To LastRowCAR

            If Len(Trim(wsPrint.Cells(i, "E").Value)) > 0 Then

                wsPrint.Cells(i, "E").Value = _
                    CleanBranch(wsPrint.Cells(i, "E").Value)

            End If

        Next i

    End If


    '========================================================
    ' STEP 13 - Generate CAR numbering dynamically
    ' B = #
    '========================================================

    If LastRowCAR >= 2 Then

        wsPrint.Range("B2:B" & LastRowCAR).Value = _
            wsPrint.Evaluate("ROW(1:" & LastRowCAR - 1 & ")")

    End If


    '========================================================
    ' STEP 14 - Format CAR area
    '========================================================
    
    If LastRowCAR >= 2 Then
    
        With wsPrint.Range("B2:F" & LastRowCAR)
    
            With .Font
                .Name = "Arial"
                .Size = 18
                .Bold = True
                .Strikethrough = False
                .Superscript = False
                .Subscript = False
                .OutlineFont = False
                .Shadow = False
                .Underline = xlUnderlineStyleNone
                .Color = -16777216
                .TintAndShade = 0
                .ThemeFont = xlThemeFontNone
            End With
    
            With .Borders
                .LineStyle = xlContinuous
                .Weight = xlThin
            End With
    
        End With
    
    End If
    
    
    '========================================================
    ' STEP 15 - CLEAN BRANCHES + AUTOFIT + FINISH
    '========================================================
    
    CleanBranchesAndFinish wsPrint, LastRowCAR, LastRowLME

End Sub


'============================================================
' CLEAN BRANCHES + AUTOFIT + FINISH
'============================================================

Private Sub CleanBranchesAndFinish( _
    ByVal wsPrint As Worksheet, _
    ByVal LastRowCAR As Long, _
    ByVal LastRowLME As Long)

    Dim i As Long


    '========================================================
    ' CLEAN CAR BRANCHES
    ' CAR Branch = Column F
    '========================================================

    If LastRowCAR >= 2 Then

        For i = 2 To LastRowCAR

            If Len(Trim(wsPrint.Cells(i, "F").Value)) > 0 Then

                wsPrint.Cells(i, "F").Value = _
                    CleanBranch(wsPrint.Cells(i, "F").Value)

            End If

        Next i

    End If


    '========================================================
    ' CLEAN LME BRANCHES
    ' LME Branch = Column K
    '========================================================

    If LastRowLME >= 2 Then

        For i = 2 To LastRowLME

            If Len(Trim(wsPrint.Cells(i, "K").Value)) > 0 Then

                wsPrint.Cells(i, "K").Value = _
                    CleanBranch(wsPrint.Cells(i, "K").Value)

            End If

        Next i

    End If


    '========================================================
    ' AutoFit AFTER cleaning all branch names
    '========================================================

    wsPrint.Cells.EntireColumn.AutoFit


    '========================================================
    ' Clear clipboard
    '========================================================

    Application.CutCopyMode = False


    '========================================================
    ' FINISH
    '========================================================

    ThisWorkbook.Activate
    ThisWorkbook.Worksheets("Sheet1").Select
    ThisWorkbook.Worksheets("Sheet1").Range("A1").Select

End Sub


'============================================================
' CLEAN BRANCH FUNCTION
'============================================================

Private Function CleanBranch(ByVal BranchName As String) As String

    Dim Branch As String

    '--------------------------------------------------------
    ' Remove spaces before and after the branch name
    '--------------------------------------------------------

    Branch = Trim(BranchName)


    '--------------------------------------------------------
    ' Branch mapping
    '--------------------------------------------------------

    Select Case Branch

        Case "Sodic Strip Mall Branch"
            CleanBranch = "Sodic"

        Case "Emad ElDin Branch"
            CleanBranch = "Emad ElDin"

        Case "Cairo Festival City"
            CleanBranch = "Cairo Festival"

        Case "Lebanon Street"
            CleanBranch = "Lebanon"

        Case "Heliopolis Al andalous"
            CleanBranch = "Al andalous"

        Case "Heliopolis Meriland"
            CleanBranch = "Meriland"

        Case "New Maadi - Road 9"
            CleanBranch = "New Maadi"

        Case "Korba Branch"
            CleanBranch = "Korba"

        Case "ElObour City Branch"
            CleanBranch = "ElObour"

        Case "kafr El Sheikh Elgaish street"
            CleanBranch = "kafr El Sheikh"

        Case "Zagazig Elgalaa street"
            CleanBranch = "Zagazig Elgalaa"

        Case "Tanta Elgaish street"
            CleanBranch = "Tanta Elgaish"

        Case "ElRehab Branch"
            CleanBranch = "ElRehab"

        Case "ElThawra str Branch"
            CleanBranch = "ElThawra"

        Case "Sohag Nile branch"
            CleanBranch = "Sohag Nile"

        Case "Sheraton Branch"
            CleanBranch = "Sheraton"

        Case "Hurghada Branch"
            CleanBranch = "Hurghada"

        Case "Madinaty Branch"
            CleanBranch = "Madinaty"

        Case "New Cairo Branch"
            CleanBranch = "New Cairo"

        Case "El Shams Branch"
            CleanBranch = "El Shams"

        Case "Arabella Branch"
            CleanBranch = "Arabella"

        Case "Salah Salem Branch"
            CleanBranch = "Salah Salem"

        Case "Assiut Gomhouria Street"
            CleanBranch = "Assiut Gomhouria"

        Case "Minia - korrniash bransh"
            CleanBranch = "Minia - korrniash"

        Case "Shoubra Branch"
            CleanBranch = "Shoubra"

        Case "Sohag"
            CleanBranch = "Sohag"

        '----------------------------------------------------
        ' Unknown branch:
        ' Keep the original name but remove spaces
        ' before and after it.
        '----------------------------------------------------

        Case Else
            CleanBranch = Trim(Branch)

    End Select

End Function
