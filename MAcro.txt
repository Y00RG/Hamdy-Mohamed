Sub Automatic_BO()
'
' Automatic_BO Macro
'

'
    Windows("BO.xlsx").Activate
    Cells.Select
    Cells.EntireColumn.AutoFit
    Columns("A:K").Select
    Selection.Delete Shift:=xlToLeft
    Columns("D:E").Select
    Selection.Delete Shift:=xlToLeft
    Columns("G:AO").Select
    Selection.Delete Shift:=xlToLeft
    Range("G1").Select
    Selection.End(xlToLeft).Select
    Selection.End(xlUp).Select
    Range("A2").Select
    Range(Selection, Selection.End(xlToRight)).Select
    Range(Selection, Selection.End(xlDown)).Select
    Selection.Copy
    Windows("BO report ( Setteled cases ).xlsx").Activate
    Range("A1").Select
    Selection.End(xlDown).Select
    Range("A166102").Select
    ActiveSheet.Paste
    ActiveWindow.SmallScroll Down:=6
    Application.CutCopyMode = False
    ActiveWorkbook.Save
    Windows("BO.xlsx").Activate
    Range("G84").Select
    Selection.End(xlUp).Select
    ActiveSheet.Range("$A$1:$G$177").AutoFilter Field:=7, Criteria1:= _
        "=Rejected", Operator:=xlOr, Criteria2:="="
    Rows("7:852").Select
    Selection.Delete Shift:=xlUp
    ActiveSheet.Range("$A$1:$G$76").AutoFilter Field:=7
    Columns("E:E").Select
    Selection.Cut
    Columns("D:D").Select
    Selection.Insert Shift:=xlToRight
    Columns("B:B").Select
    Selection.Cut
    Columns("G:G").Select
    Selection.Insert Shift:=xlToRight
    Range("C2").Select
    Range(Selection, Selection.End(xlDown)).Select
    Selection.ClearContents
    Range("A1").Select
 
    ActiveSheet.Range("$A$1:$G$76").AutoFilter Field:=1, Criteria1:=Array("LME", "LSG", "LSC", "LUG", "LSH", "LSE"), Operator:=xlFilterValues
    
    Range("A1").Select
    Selection.End(xlUp).Select
    Range(Selection, Selection.End(xlToRight)).Select
    Range(Selection, Selection.End(xlDown)).Select
    Selection.Copy
    Sheets.Add After:=ActiveSheet
    Sheets("Sheet1").Select
    Sheets("Sheet1").Name = "LME"
    ActiveSheet.Paste
    Cells.Select
    Cells.EntireColumn.AutoFit
    ActiveWindow.Zoom = 115
    ActiveWindow.Zoom = 145
    ActiveWindow.Zoom = 160
    ActiveWindow.Zoom = 175
    Range("C2").Select
    Sheets("Report 1").Select
    ActiveWindow.SmallScroll Down:=-2
    Range("A1").Select
    ActiveSheet.Range("$A$1:$G$76").AutoFilter Field:=1
    
    ActiveSheet.Range("$A$1:$G$76").AutoFilter Field:=1, Criteria1:=Array("MCF", "UMC", "MGS"), Operator:=xlFilterValues
    
    Range("A1").Select
    Range(Selection, Selection.End(xlToRight)).Select
    Range(Selection, Selection.End(xlDown)).Select
    Application.CutCopyMode = False
    Selection.Copy
    Sheets.Add After:=ActiveSheet
    Sheets("Sheet2").Select
    Sheets("Sheet2").Name = "CAR"
    Range("A1").Select
    ActiveSheet.Paste
    Cells.Select
    Cells.EntireColumn.AutoFit
    ActiveWindow.Zoom = 115
    ActiveWindow.Zoom = 130
    ActiveWindow.Zoom = 145
    ActiveWindow.Zoom = 160
    Range("C2").Select
    Application.CutCopyMode = False
    Application.CutCopyMode = False
    Application.CutCopyMode = False
    ActiveCell.FormulaR1C1 = "=VLOOKUP(D2,'[BO . Coding.xlsx]Car'!$A:$B,2,0)"
    Selection.TextToColumns Destination:=Range("C2"), DataType:=xlDelimited, _
        TextQualifier:=xlDoubleQuote, ConsecutiveDelimiter:=False, Tab:=True, _
        Semicolon:=False, Comma:=False, Space:=False, Other:=False, FieldInfo _
        :=Array(Array(1, 1), Array(2, 1), Array(3, 1)), TrailingMinusNumbers:=True

    Dim LastRow As Long
    LastRow = Cells(Rows.Count, "D").End(xlUp).Row
    Range("C2").AutoFill Destination:=Range("C2:C" & LastRow)
    
    Columns("C:C").Select
    
    ActiveWorkbook.Worksheets("CAR").Sort.SortFields.Clear
    ActiveWorkbook.Worksheets("CAR").Sort.SortFields.Add2 Key:=Range("C1:C44"), _
        SortOn:=xlSortOnValues, Order:=xlAscending, DataOption:=xlSortNormal
    With ActiveWorkbook.Worksheets("CAR").Sort
        .SetRange Range("A2:G44")
        .Header = xlNo
        .MatchCase = False
        .Orientation = xlTopToBottom
        .SortMethod = xlPinYin
        .Apply
    End With
    Range("C2").Select
    Sheets("LME").Select
    Range("C2").Select
    Application.CutCopyMode = False
    ActiveCell.FormulaR1C1 = "=VLOOKUP(D2,'[BO . Coding.xlsx]LME'!$A:$B,2,0)"
    Selection.TextToColumns Destination:=Range("C2"), DataType:=xlDelimited, _
        TextQualifier:=xlDoubleQuote, ConsecutiveDelimiter:=False, Tab:=True, _
        Semicolon:=False, Comma:=False, Space:=False, Other:=False, FieldInfo _
        :=Array(Array(1, 1), Array(2, 1), Array(3, 1)), TrailingMinusNumbers:=True
    
    Dim LastRow As Long
    LastRow = Cells(Rows.Count, "D").End(xlUp).Row
    Range("C2").AutoFill Destination:=Range("C2:C" & LastRow)
    
    Columns("C:C").Select
    
    ActiveWorkbook.Worksheets("LME").Sort.SortFields.Clear
    ActiveWorkbook.Worksheets("LME").Sort.SortFields.Add2 Key:=Range("C1:C33"), _
        SortOn:=xlSortOnValues, Order:=xlAscending, DataOption:=xlSortNormal
    With ActiveWorkbook.Worksheets("LME").Sort
        .SetRange Range("A2:G33")
        .Header = xlNo
        .MatchCase = False
        .Orientation = xlTopToBottom
        .SortMethod = xlPinYin
        .Apply
    End With
    Range("C2").Select
    Windows("Automatic_BO.xlsm").Activate
End Sub
